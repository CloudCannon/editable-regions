{{- /* Recursively walks a directory, returning a dict of path -> file contents for files matching the extension list. Expects: (dict "dir" "layouts/partials" "exts" (slice ".html" ".htm")) Optional: "keyDir" — when set, dict keys are "keyDir/" instead of "dir/", letting the caller read from a physical module directory (e.g. "themes/proot/layouts/partials") while keying the snapshot under the logical layout path (e.g. "layouts/partials"). Note: readDir/readFile operate on the project directory (plus the union overlay for known paths), so physical module/theme paths enumerate fine while logical paths do NOT — hence the keyDir remap for theme/vendor walking. */ -}} {{- $dir := .dir -}} {{- $exts := .exts -}} {{- $keyDir := .keyDir | default $dir -}} {{- $files := dict -}} {{- if os.FileExists $dir -}} {{- range $entry := readDir $dir -}} {{- $path := printf "%s/%s" $dir $entry.Name -}} {{- $key := printf "%s/%s" $keyDir $entry.Name -}} {{- if $entry.IsDir -}} {{- $files = merge $files (partial "editable-regions/walk-dir.html" (dict "dir" $path "keyDir" $key "exts" $exts)) -}} {{- else -}} {{- $ext := path.Ext $entry.Name | lower -}} {{- if in $exts $ext -}} {{- $files = merge $files (dict $key (readFile $path)) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $files -}}
