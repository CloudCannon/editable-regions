{{- $cfg := site.Params.editable_regions | default dict -}} {{- $templateExts := $cfg.template_extensions | default (slice ".html" ".htm") -}} {{- $ignoreDirs := $cfg.ignore_directories | default (slice ".git" "node_modules" "public" "resources") -}} {{- $overrides := $cfg.templates_overrides | default dict -}} {{- $wasmUrl := $cfg.wasm_url -}} {{- if not $wasmUrl -}} {{- $wasmUrl = partialCached "editable-regions/wasm-url.html" (dict "override_version" ($cfg._version | default "") "base_url" ($cfg.wasm_base_url | default "https://github.com/CloudCannon/editable-regions/releases/download") ) -}} {{- end -}} {{- /* Template snapshot: explicit template_dirs replace the recursive project walk (and its theme/vendored-module discovery) entirely — each listed dir is walked verbatim and keyed at its given paths. Otherwise the usual discovery captures partials, render hooks, and shortcodes from the project, theme, and vendored modules. */ -}} {{- $templateFiles := dict -}} {{- if $cfg.template_dirs -}} {{- range $td := $cfg.template_dirs -}} {{- $templateFiles = merge $templateFiles (partial "editable-regions/find-files-with-extension.html" (dict "dir" $td "exts" $templateExts "ignore" $ignoreDirs)) -}} {{- end -}} {{- else -}} {{- $deps := partial "editable-regions/load-deps.html" -}} {{- $templateFiles = partial "editable-regions/find-template-files.html" (dict "deps" $deps "exts" $templateExts "ignore" $ignoreDirs) -}} {{- /* Vendored modules resolve from _vendor/ only when modules.txt (the vendor manifest) exists, so mirror it verbatim. */ -}} {{- if os.FileExists "_vendor/modules.txt" -}} {{- $templateFiles = merge $templateFiles (dict "_vendor/modules.txt" (readFile "_vendor/modules.txt")) -}} {{- end -}} {{- end -}} {{- /* Site config captured at build time so the editor loads the real config (theme, module imports, params, menus, …) through Hugo's own resolution. Explicit config_paths skip discovery and capture only the listed files, verbatim. */ -}} {{- $configFiles := dict -}} {{- if $cfg.config_paths -}} {{- range $cp := $cfg.config_paths -}} {{- if os.FileExists $cp -}} {{- $configFiles = merge $configFiles (dict $cp (readFile $cp)) -}} {{- end -}} {{- end -}} {{- else -}} {{- $configFiles = partial "editable-regions/find-config-files.html" -}} {{- end -}} {{- /* Template overrides: name -> physical source path, mirrored verbatim (keyed at that path) so the renderer has the exact file; it maps the name to a reserved partial at runtime so the override renders regardless of lookup/collision order. */ -}} {{- if $overrides -}} {{- range $name, $overridePath := $overrides -}} {{- if os.FileExists $overridePath -}} {{- $templateFiles = merge $templateFiles (dict $overridePath (readFile $overridePath)) -}} {{- end -}} {{- end -}} {{- end -}} {{- $ctx := dict "meta" (dict "generator" (printf "Hugo %s" hugo.Version) "wasmUrl" $wasmUrl "verbose" ($cfg.verbose | default false) "env" (hugo.Environment | default "production") "templateOverrides" $overrides ) "files" (merge $templateFiles $configFiles) -}} {{- $entry := resources.Get "integrations/hugo/browser/entry.js" -}} {{- if not $entry -}} {{- errorf "editable-regions: integrations/hugo/browser/entry.js not found in assets — is the editable-regions module imported and mounted?" -}} {{- end -}} {{- return ($entry | resources.ExecuteAsTemplate "cc-editable-regions/live-editing.js" $ctx | js.Build (dict "minify" (not hugo.IsDevelopment)) | fingerprint) -}}
