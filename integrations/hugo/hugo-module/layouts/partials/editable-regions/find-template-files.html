{{- $files := dict -}} {{- $subDirs := slice "partials" "shortcodes" -}} {{- $deps := .deps -}} {{- $exts := .exts -}} {{- $ignore := .ignore | default slice -}} {{- $dir := .dir | default "." -}} {{- if os.FileExists $dir -}} {{- range $entry := readDir $dir -}} {{- if not $entry.IsDir -}} {{ continue }} {{- end -}} {{- $path := path.Join $dir $entry.Name -}} {{- range $dep := $deps -}} {{- $resolved := path.Clean (path.Join $dir $dep) -}} {{- if and (not (strings.HasPrefix $resolved "..")) (os.FileExists $resolved) -}} {{- $files = merge $files (partial "editable-regions/find-dep-template-files.html" (dict "dir" $resolved "exts" $exts "ignore" $ignore)) -}} {{- end -}} {{- end -}} {{- $isDep := false -}} {{- range $dep := $deps -}} {{- if strings.HasSuffix $path (printf "/%s" $dep) -}} {{- $files = merge $files (partial "editable-regions/find-dep-template-files.html" (dict "dir" $path "exts" $exts "ignore" $ignore)) -}} {{- $isDep = true -}} {{- break -}} {{- end -}} {{- end -}} {{- if or $isDep (in $ignore $entry.Name) -}} {{ continue }} {{- end -}} {{- if in $subDirs $entry.Name -}} {{- $files = merge $files (partial "editable-regions/find-files-with-extension.html" (dict "dir" $path "exts" $exts "ignore" $ignore)) -}} {{- else if eq $entry.Name "_default" -}} {{- range $defaultEntry := readDir $path -}} {{- if eq $defaultEntry.Name "_markup" -}} {{- $files = merge $files (partial "editable-regions/find-files-with-extension.html" (dict "dir" (path.Join $path "_markup") "exts" $exts "ignore" $ignore)) -}} {{- end -}} {{- end -}} {{- end -}} {{- $files = merge $files (partial "editable-regions/find-template-files.html" (dict "dir" $path "deps" $deps "exts" $exts "ignore" $ignore)) -}} {{- end -}} {{- end -}} {{- return $files -}}
