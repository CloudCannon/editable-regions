{{- $dir := .dir -}} {{- $exts := .exts -}} {{- $ignore := .ignore | default slice -}} {{- /* Config extensions double as the data/i18n file extensions. */ -}} {{- $configExts := slice ".toml" ".yaml" ".yml" ".json" -}} {{- $files := dict -}} {{- /* Dep root config files, captured verbatim at their physical paths so the renderer's module resolution sees them. */ -}} {{- range $candidate := slice "hugo.toml" "hugo.yaml" "hugo.yml" "hugo.json" "config.toml" "config.yaml" "config.yml" "config.json" -}} {{- $cfgPath := path.Join $dir $candidate -}} {{- if os.FileExists $cfgPath -}} {{- $files = merge $files (dict $cfgPath (readFile $cfgPath)) -}} {{- end -}} {{- end -}} {{- /* The dep's own config, parsed for module.mounts: hugo.* first, then config.* (Hugo's root precedence). */ -}} {{- $conf := dict -}} {{- range $candidate := slice "hugo.toml" "hugo.yaml" "hugo.yml" "hugo.json" "config.toml" "config.yaml" "config.yml" "config.json" -}} {{- $cfgPath := path.Join $dir $candidate -}} {{- if os.FileExists $cfgPath -}} {{- $ext := strings.TrimPrefix "." (path.Ext $candidate) -}} {{- $conf = os.ReadFile $cfgPath | transform.Unmarshal (dict "format" $ext) -}} {{- break -}} {{- end -}} {{- end -}} {{- $mounts := slice -}} {{- with $conf.module -}} {{- with .mounts -}} {{- $mounts = . -}} {{- end -}} {{- end -}} {{- /* Walks to run, handed straight to find-files-with-extension (which no-ops on missing dirs). Files are captured at their physical paths — the renderer's module resolution re-derives these mounts from the dep config captured above, so no target mapping is needed here. Layouts sources are walked whole: pages build as render:link and the dispatch view is looked up by exact path, so kind layouts and baseofs never execute — they are inert but must parse under the renderer's Hugo. */ -}} {{- $walks := slice -}} {{- if eq (len $mounts) 0 -}} {{- /* No mounts: Hugo's implicit module trees. */ -}} {{- $walks = $walks | append (dict "dir" (path.Join $dir "layouts") "exts" $exts "ignore" $ignore) (dict "dir" (path.Join $dir "i18n") "exts" $configExts) (dict "dir" (path.Join $dir "data") "exts" $configExts) -}} {{- else -}} {{- range $mount := $mounts -}} {{- $source := $mount.source | default "" -}} {{- if eq $source "." -}} {{- $source = "" -}} {{- end -}} {{- $target := $mount.target | default "" -}} {{- $src := path.Join $dir $source -}} {{- if or (eq $target "layouts") (strings.HasPrefix $target "layouts/") -}} {{- $walks = $walks | append (dict "dir" $src "exts" $exts "ignore" $ignore) -}} {{- else if or (eq $target "i18n") (strings.HasPrefix $target "i18n/") (eq $target "data") (strings.HasPrefix $target "data/") -}} {{- $walks = $walks | append (dict "dir" $src "exts" $configExts) -}} {{- end -}} {{- /* Other targets (content/static/assets/archetypes) stay uncaptured: content comes from the CloudCannon mirror, the rest is unused by the editor. */ -}} {{- end -}} {{- end -}} {{- range $walk := $walks -}} {{- $files = merge $files (partial "editable-regions/find-files-with-extension.html" $walk) -}} {{- end -}} {{- return $files -}}
