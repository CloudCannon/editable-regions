{{- /* Recursively walks a directory and returns every file whose extension matches "exts", keyed by its physical (path-joined) path; callers merge the results. */ -}} {{- $dir := .dir -}} {{- $exts := .exts -}} {{- $files := dict -}} {{- if os.FileExists $dir -}} {{- range $entry := readDir $dir -}} {{- $path := path.Join $dir $entry.Name -}} {{- if $entry.IsDir -}} {{- $files = merge $files (partial "editable-regions/find-files-with-extension.html" (dict "dir" $path "exts" $exts)) -}} {{- else -}} {{- if in $exts (path.Ext $entry.Name | lower) -}} {{- $files = merge $files (dict $path (readFile $path)) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $files -}}
