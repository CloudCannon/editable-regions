{{- /* Serializes the browser-relevant slice of the site configuration. The runtime writes this (plus editor overrides) as the WASM site's config, so component partials can use `site.Params`, `site.Title`, and `site.Menus`. No directory keys are forwarded: the editor site always uses Hugo's default layoutDir/dataDir/contentDir. The template snapshot is keyed under the canonical layouts/ root (walk-project.html), and content/data are mirrored from CloudCannon collections/datasets under content//data/ at runtime (see the runtime). Provisioned by editable-regions/resources.html. */ -}} {{- $menus := dict -}} {{- range $name, $entries := site.Menus -}} {{- $items := slice -}} {{- range $entries -}} {{- $items = $items | append (dict "name" .Name "url" .URL "weight" .Weight) -}} {{- end -}} {{- $menus = merge $menus (dict $name $items) -}} {{- end -}} {{- return (dict "baseURL" ("/" | absURL) "title" site.Title "locale" (site.LanguageCode | default "") "params" site.Params "menus" $menus ) -}}
