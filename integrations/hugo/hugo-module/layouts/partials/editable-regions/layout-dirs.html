{{- /* Detects the site's configured directory base names from its config file(s). Hugo templates can't read config directly, so we parse the config file(s) themselves. Mirrors Hugo's lookup order (config/configLoader.go): root "hugo.*" names before legacy "config.*", and root files before config/_default/*. First-found-wins is an approximation — Hugo merges the config dir over the root file per key, so a key set only in config/_default/ (with a root config present) is missed. Values fall back to Hugo's defaults when unset. Returns: (dict "layoutDir" "dataDir" "contentDir"). The os.* funcs read Hugo's union filesystem (the project layered over every imported module's root), so a candidate name may resolve to a MODULE's own config — e.g. an imported module shipping a root hugo.toml appears at the union path "hugo.toml" when the project itself uses "config.toml". Candidates are therefore accepted only when they carry a site-level fingerprint (title, baseURL, params, menus, dir keys, or module imports/replacements); module configs (typically just [module] with mounts) are skipped so the probe falls through to the project's real config. */ -}} {{- $layoutDir := "layouts" -}} {{- $dataDir := "data" -}} {{- $contentDir := "content" -}} {{- $candidates := slice "hugo.toml" "hugo.yaml" "hugo.yml" "hugo.json" "config.toml" "config.yaml" "config.yml" "config.json" "config/_default/hugo.toml" "config/_default/hugo.yaml" "config/_default/hugo.yml" "config/_default/hugo.json" -}} {{- range $file := $candidates -}} {{- if os.FileExists $file -}} {{- $ext := strings.TrimPrefix "." (path.Ext $file) -}} {{/* TrimPrefix is (prefix, s) — prefix first */}} {{- $conf := os.ReadFile $file | transform.Unmarshal (dict "format" $ext) -}} {{- $siteLevel := false -}} {{- if or (isset $conf "layoutDir") (isset $conf "dataDir") (isset $conf "contentDir") (isset $conf "themesDir") (isset $conf "theme") (isset $conf "baseURL") (isset $conf "title") (isset $conf "languageCode") (isset $conf "params") (isset $conf "menu") (isset $conf "menus") -}} {{- $siteLevel = true -}} {{- else if isset $conf "module" -}} {{- with $conf.module.imports -}} {{- $siteLevel = true -}} {{- end -}} {{- with $conf.module.replacements -}} {{- $siteLevel = true -}} {{- end -}} {{- end -}} {{- if $siteLevel -}} {{- with $conf.layoutDir -}}{{- $layoutDir = . -}}{{- end -}} {{- with $conf.dataDir -}}{{- $dataDir = . -}}{{- end -}} {{- with $conf.contentDir -}}{{- $contentDir = . -}}{{- end -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "layoutDir" $layoutDir "dataDir" $dataDir "contentDir" $contentDir) -}}
