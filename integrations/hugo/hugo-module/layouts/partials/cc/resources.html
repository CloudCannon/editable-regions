{{- /* Builds the single live-editing bundle: the site snapshot prelude (cc/snapshot.html) concatenated with the prebuilt runtime, fingerprinted for cache-busting. The snapshot comes first — the runtime reads the window.cc_hugo* globals it assigns. Returns the resource. cc/live-editing-head.html is the batteries-included wrapper; call this directly if you need control over the script tag (conditional loading, your own pipeline, etc.). */ -}} {{- $snapshot := partial "cc/snapshot.html" . -}} {{- $data := resources.FromString "cc-editable-regions/snapshot.js" $snapshot -}} {{- $runtime := resources.Get "cc-editable-regions/runtime.js" -}} {{- if not $runtime -}} {{- errorf "editable-regions: cc-editable-regions/runtime.js not found in assets (build it with integrations/hugo/build-runtime.mjs)" -}} {{- end -}} {{- return (resources.Concat "cc-editable-regions/live-editing.js" (slice $data $runtime) | fingerprint) -}}
